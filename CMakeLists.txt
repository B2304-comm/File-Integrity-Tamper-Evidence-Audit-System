cmake_minimum_required(VERSION 3.16)
project(FileIntegrityAudit VERSION 1.0.0 LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "This project uses Linux system calls. Build in Linux or Ubuntu through WSL.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
add_library(audit_core
  src/HashManager.cpp src/FileScanner.cpp src/IntegrityChecker.cpp
  src/AuditLogger.cpp src/ConfigManager.cpp src/FileMonitor.cpp)
target_include_directories(audit_core PUBLIC include)
target_link_libraries(audit_core PUBLIC OpenSSL::Crypto)
target_compile_options(audit_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
add_executable(audit src/main.cpp)
target_link_libraries(audit PRIVATE audit_core)
target_compile_options(audit PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
  add_executable(unit_tests tests/unit_tests.cpp)
  target_link_libraries(unit_tests PRIVATE audit_core)
  add_test(NAME unit COMMAND unit_tests)
  add_executable(integration_tests tests/integration_tests.cpp)
  target_link_libraries(integration_tests PRIVATE OpenSSL::Crypto)
  add_dependencies(integration_tests audit)
  add_test(NAME integration COMMAND integration_tests $<TARGET_FILE:audit>)
  set_tests_properties(integration PROPERTIES TIMEOUT 60)
endif()
install(TARGETS audit RUNTIME DESTINATION bin)
